function [D,Reference,SignedFraction]=buildBOIBaselineDiagnostic(Data,fs)
%BUILDBOIBASELINEDIAGNOSTIC Review-only reference; never replaces a measurement.
F=Data.Frames;row=Data.Row;raw=F.PreservedInputMean;N=height(F);
Reference=nan(N,1);SignedFraction=nan(N,1);b=find(F.CleanBaseline);event=find(F.MeasurementWindow);
D=struct('Schema','boi-baseline-diagnostic-1','MethodID','boi-baseline-linear-review-1', ...
    'Role','review_only_not_saved_measurement','Status','unavailable','FitAvailable',false, ...
    'RecordingID',char(row.RecordingID),'EventType',char(row.EventType),'SiteID',row.SiteID,'EventID',row.EventID, ...
    'RawSourceSHA256',char(row.SourceRawSHA256),'SavedAmplitudeFraction',row.StoredAmplitude, ...
    'Source','preserved_input_fixed_native_union','AmplitudeUnits','fraction; percent = 100 * fraction', ...
    'SignedAUCUnits','fraction_seconds','ComparisonUnits','percentage_points', ...
    'Message','Diagnostic unavailable.','ScientificStatus','not_established', ...
    'ReferenceFormula','mean(y) + sum((t-mean(t)).*(y-mean(y)))/sum((t-mean(t)).^2) * (time-mean(t))', ...
    'AmplitudeFormula','sink: -min((source-reference)/reference); surge: max((source-reference)/reference)', ...
    'SampleF',fs,'RequiredSamples',row.BaselineWindowFrames,'CleanSamples',numel(b), ...
    'BaselineDurationSec',row.BaselineWindowFrames/fs,'BaselineStartFrame',NaN,'BaselineEndFrame',NaN, ...
    'FirstHalfSamples',0,'LastHalfSamples',0,'FootprintPixels',numel(Data.Footprint), ...
    'BaselineMean',NaN,'SlopeUnitsPerSec',NaN,'SlopePercentPerSec',NaN, ...
    'FirstHalfSlopePercentPerSec',NaN,'LastHalfSlopePercentPerSec',NaN, ...
    'CVPercent',NaN,'PeakToTroughPercent',NaN,'ResidualRMSPercent',NaN,'DescriptiveR2',NaN, ...
    'MeasurementStartFrame',row.StartFrame,'MeasurementEndFrame',row.EndFrame, ...
    'ExtrapolationDurationSec',numel(event)/fs,'FirstEventReference',NaN,'LastEventReference',NaN, ...
    'MinimumEventReference',NaN,'AmplitudeFraction',NaN,'MeanSignedChangeFraction',NaN, ...
    'SignedAUC_sec',NaN,'DifferenceFromSavedAmplitude_pp',NaN, ...
    'StabilityDecision','not_assigned', ...
    'Limitations','A short fluctuation can be extrapolated incorrectly. No substrate kinetics, correction or automatic exclusion is inferred.', ...
    'NumericalInterpretation','Negative and very small values are retained; numerical sign alone is not physiological evidence. No sign cutoff is applied.');
n=D.RequiredSamples;
if ~isscalar(n)||~isfinite(n)||n<1||n~=fix(n)||numel(b)~=n|| ...
        ~isequal(b,(row.StartFrame-n:row.StartFrame-1)')
    D.Status='incomplete_or_nonlocal_baseline';
    D.Message='The diagnostic requires every sample in the saved immediate pre-event baseline. No earlier samples are substituted.';return;
end
if any(~isfinite(raw(b)))
    D.Status='nonfinite_baseline';D.Message='At least one required baseline source sample is nonfinite.';return;
end
if row.RecomputedStatus~="valid"
    D.Status='saved_baseline_unavailable';D.Message=['Saved baseline status: ' char(row.RecomputedStatus) '. The diagnostic does not repair it.'];return;
end
if ~Data.BaselineReplayMatches||(~Data.ReplayMatches&&all(isfinite(raw(event))))
    D.Status='saved_trace_replay_mismatch';D.Message='Saved source replay disagrees with the audit. Reopen or investigate the audit before interpreting a diagnostic.';return;
end
y=raw(b);B=mean(y);
if ~isfinite(B)||B<=0
    D.Status='nonpositive_baseline';D.Message='The original-source baseline mean is nonpositive.';return;
end
if n<2
    D.Status='insufficient_samples_for_line';D.Message='A line needs at least two baseline samples. The saved measurement is retained.';return;
end
time=F.ModeledTimeSec;x=time(b)-mean(time(b));beta=sum(x.*(y-B))/sum(x.^2);
D.FitAvailable=true;D.BaselineStartFrame=b(1);D.BaselineEndFrame=b(end);D.BaselineMean=B;
D.SlopeUnitsPerSec=beta;D.SlopePercentPerSec=100*beta/B;
half=floor(n/2);D.FirstHalfSamples=half;D.LastHalfSamples=n-half;
D.FirstHalfSlopePercentPerSec=halfSlope(time(b(1:half)),y(1:half),B);
D.LastHalfSlopePercentPerSec=halfSlope(time(b(half+1:end)),y(half+1:end),B);
fit=B+beta*x;D.CVPercent=100*std(y)/B;D.PeakToTroughPercent=100*(max(y)-min(y))/B;
D.ResidualRMSPercent=100*sqrt(mean((y-fit).^2))/B;
variance=sum((y-B).^2);if variance>0,D.DescriptiveR2=1-sum((y-fit).^2)/variance;end
use=[b;event];Reference(use)=B+beta*(time(use)-mean(time(b)));
D.FirstEventReference=Reference(event(1));D.LastEventReference=Reference(event(end));
D.MinimumEventReference=min(Reference(event));
if any(~isfinite(raw(event)))||any(~isfinite(Reference(use)))
    D.Status='nonfinite_event_source_or_reference';D.Message='The fitted line is shown for inspection; the alternative amplitude is unavailable because event source or reference values are nonfinite.';return;
end
if any(Reference(event)<=0)
    D.Status='nonpositive_event_reference';D.Message='The extrapolated line reaches a nonpositive event reference. Its amplitude and integral are unavailable; no clipping or fallback is applied.';return;
end
SignedFraction(event)=(raw(event)-Reference(event))./Reference(event);
q=SignedFraction(event);if row.EventType=="sink",D.AmplitudeFraction=-min(q);else,D.AmplitudeFraction=max(q);end
D.MeanSignedChangeFraction=mean(q);D.SignedAUC_sec=sum(q)/fs;
D.DifferenceFromSavedAmplitude_pp=100*(D.AmplitudeFraction-row.StoredAmplitude);
D.Status='available_diagnostic_only';D.Message='Calculated for review only. The saved amplitude, baseline and event windows remain unchanged.';
end
function beta=halfSlope(t,y,B)
beta=NaN;if numel(t)<2,return;end
x=t-mean(t);beta=100*sum(x.*(y-mean(y)))/sum(x.^2)/B;
end
