function [Timing,F]=buildBOITimingReviewData(t,I,Review)
%BUILDBOITIMINGREVIEWDATA Expose saved stages without refitting or substitution.
N=I.NFrames;
[corrected,C]=stage(t,'DetectionDetrended',N,'primary_timing_reference','corrected intensity', ...
    'fixed_event_native_union','Saved per-pixel-corrected signal averaged over the event footprint.');
[filtered,S]=stage(t,'Filtered',N,'supporting_detection_score','detection score', ...
    'fixed_event_native_union','Saved normalized and spatially/temporally filtered detection signal.');
[site,U]=stage(t,'TimingTrace',N,'supporting_site_context','site score', ...
    'saved_site_support_not_event_footprint','Saved TimingTrace: filtered site trace for sinks, normalized site trace for surges. No TimingTrend subtraction; support and processing differ.');
removed=nan(N,1);T=struct('Status','unavailable','Message','Removed trend unavailable: the correction input relationship was not saved.');
if isfield(I,'DenoisedFile')
    if ~isempty(I.DenoisedFile)
        T.Message='Removed trend unavailable: correction used a different detection input from the preserved raw source.';
    elseif C.ShapeValid
        removed=double(t.Raw(:))-corrected;T.Status='derived_from_saved_stages';
        T.Message='Already removed trend = preserved raw mean minus saved corrected mean on the same event footprint. No new fit.';
        if ~any(isfinite(removed)),T.Status='no_finite_samples';end
    else
        T.Message=['Removed trend unavailable: ' C.Message];
    end
end
Timing=struct('Schema','boi-timing-review-1','Role','saved_evidence_review_only', ...
    'PrimaryReference','CorrectedIntensity','SupportingReferences',{{'FilteredDetectionScore','SavedSiteTimingTrace'}}, ...
    'RawRole','correction_quality_check; amplitude remains in the saved measurement view', ...
    'Corrected',C,'Filtered',S,'Site',U,'RemovedTrend',T, ...
    'SavedAuditPath',Review.AuditPath,'SavedAuditSHA256',Review.AuditSHA256, ...
    'RawSourceSHA256',I.RawSHA256,'FootprintPixels',numel(t.Footprint), ...
    'SampleHz',I.AnalysisParams.fs,'FrameOrigin',1,'ModeledTimeOriginSec',0, ...
    'FrameTimeFormula','ModeledTimeSec = (Frame - 1) / SampleHz', ...
    'FrameTimingStatus',char(Review.Acquisition.FrameTimingStatus), ...
    'ClockRole','Saved acquisition authority retained; embedded timestamps not substituted. Frame interval is not exposure duration.', ...
    'BoundaryRole','Saved measurement and native sample indices; neither is an accepted manual physiological boundary.', ...
    'PolicyOrigin','Researcher workflow clarification, R2-CORRECTED-SCORE-SUPPORT-049; no automated timing rule adopted.', ...
    'NewFits',0,'DetectorReruns',0,'AmplitudeChanges',false);
if isfield(I,'DenoisedSHA256'),Timing.DetectionInputSHA256=I.DenoisedSHA256;end
F=table(corrected,filtered,site,removed,'VariableNames', ...
    {'CorrectedIntensity','FilteredDetectionScore','SavedSiteTimingTrace','ExistingRemovedTrend'});
end
function [y,M]=stage(t,name,N,role,units,support,processing)
y=nan(N,1);M=struct('SourceField',name,'Role',role,'Units',units,'Support',support, ...
    'Processing',processing,'Status','not_saved','Message',[name ' was not saved in this audit.'], ...
    'ShapeValid',false,'SavedSamples',0,'FiniteSamples',0);
if ~isfield(t,name)||isempty(t.(name)),return;end
v=t.(name);M.SavedSamples=numel(v);
if ~isnumeric(v)||~isreal(v)||~isvector(v)||numel(v)~=N
    M.Status='invalid_saved_stage';M.Message=sprintf('%s is not a real numeric vector of %d recording samples; no substitute is used.',name,N);return;
end
y=double(v(:));M.ShapeValid=true;M.FiniteSamples=nnz(isfinite(y));
if M.FiniteSamples==0
    M.Status='no_finite_samples';M.Message=[name ' has no finite saved samples.'];
elseif M.FiniteSamples<N
    M.Status='partial_nonfinite';M.Message=sprintf('%s retains %d nonfinite samples as gaps; no interpolation.',name,N-M.FiniteSamples);
else
    M.Status='available';M.Message=[name ' shown exactly as saved.'];
end
end
