function Receipt=exportBOIEventReview(Review,Index,OutputFolder,DictionaryPath)
%EXPORTBOIEVENTREVIEW Export saved ingredients to a NEW local review folder.
% Contains local paths; not a sanitized public export or scientific decision.
if nargin<4,DictionaryPath='';end
assert(~isfolder(OutputFolder)&&~isfile(OutputFolder),'OxygenDynamics:ReviewOutputExists','Choose a new folder; earlier review exports are preserved.');
assert(strcmp(oxygenFileSHA256(Review.AuditPath),Review.AuditSHA256), ...
    'OxygenDynamics:EventReviewChanged','Saved audit changed since opening. Reopen before exporting.');
if isfield(Review,'BoundaryReview')&&~isempty(Review.BoundaryReview)
    checked=loadBOIBoundaryReview(Review,Review.BoundaryReview.Path);
    assert(strcmp(checked.SHA256,Review.BoundaryReview.SHA256),'OxygenDynamics:BoundaryReviewChanged','Boundary review changed. Reload before exporting.');
    Review.BoundaryReview=checked;
end
Data=buildBOIEventReviewData(Review,Index);
ReviewedReferencePreview=buildBOIReviewedReferencePreview(Review,Index);
if isfield(Data,'WindowLink')
    assert(strcmp(oxygenFileSHA256(Data.WindowLink.ResultsPath),Data.WindowLink.ResultsSHA256), ...
        'OxygenDynamics:WindowReviewChanged','Linked statistics changed. Reopen the window and reconnect before exporting.');
end
native=[];
if isfield(Data,'NativeMaskAttachment'),native=validateBOINativeMaskSource(Review,Index);end
[~,dictionary,source]=getBOIMeasurementDictionary(DictionaryPath);
dictionaryText=fileread(source);dictionaryHash=oxygenFileSHA256(source);
role='current_definitions_context_only';if ~isempty(DictionaryPath),role='user_selected_saved_dictionary; run_correspondence_not_automatically_verified';end
mkdir(OutputFolder);
writetable(Data.Frames,fullfile(OutputFolder,'SelectedEventTrace.csv'));
[r,c]=ind2sub(Review.AnalysisInfo.FrameSize,Data.Footprint(:));
writetable(table(Data.Footprint(:),r,c,'VariableNames',{'MatlabLinearPixel','PixelRow','PixelColumn'}),fullfile(OutputFolder,'FixedAmplitudeFootprint.csv'));
if ~isempty(native)
    cells=native.FramePixels;counts=cellfun(@numel,cells);frame=repelem((1:numel(cells))',counts);
    pixels=vertcat(cells{:});[r,c]=ind2sub(native.FrameSize,pixels);
    writetable(table(frame,pixels,r,c,'VariableNames',{'Frame','MatlabLinearPixel','PixelRow','PixelColumn'}),fullfile(OutputFolder,'NativeEventFramePixels.csv'));
end
writeText(fullfile(OutputFolder,'BOIMeasurementDictionary.json'),dictionaryText);
writeText(fullfile(OutputFolder,'EventReview.txt'),strjoin(string(Data.Details),newline));
Receipt=struct('Schema','boi-event-review-export-7','CreatedUTC',char(datetime('now','TimeZone','UTC','Format','yyyy-MM-dd''T''HH:mm:ssXXX')), ...
    'ScientificStatus','not_established','SavedAuditPath',Review.AuditPath,'SavedAuditSHA256',Review.AuditSHA256, ...
    'SelectedEvent',table2struct(Data.Row),'ReplayAmplitude',Data.ReplayedAmplitude,'ReplayMatches',Data.ReplayMatches, ...
    'ReplayedBaseline',Data.ReplayedBaseline,'BaselineReplayMatches',Data.BaselineReplayMatches, ...
    'AnalysisInfo',Review.AnalysisInfo,'Acquisition',Review.Acquisition,'AcquisitionQC',Review.QC, ...
    'DictionaryVersion',dictionary.Version,'DictionarySHA256',dictionaryHash,'DictionaryRole',role, ...
    'NativeFrameMasks','not_present_in_saved_audit','SourceImageIncluded',false, ...
    'DetectorReruns',0,'StatisticsReruns',0,'BaselineDiagnostic',Data.BaselineDiagnostic,'TimingReview',Data.TimingReview);
Receipt.ReviewedReferencePreview=rmfield(ReviewedReferencePreview,{'Frames','Contributors'});
writeText(fullfile(OutputFolder,'ReviewedReferencePreview.json'),jsonencode(Receipt.ReviewedReferencePreview,'PrettyPrint',true));
if ~isempty(ReviewedReferencePreview.Windows)
    writetable(ReviewedReferencePreview.Frames,fullfile(OutputFolder,'ReviewedReferenceFrames.csv'));
    writetable(ReviewedReferencePreview.Contributors,fullfile(OutputFolder,'ReviewedReferenceContributors.csv'));
end
Receipt.ResearcherBoundaryReview=Data.ResearcherBoundaryReview;
writeText(fullfile(OutputFolder,'ResearcherBoundaries.json'),jsonencode(Data.ResearcherBoundaryReview,'PrettyPrint',true));
if isfield(Review,'BoundaryReview')&&~isempty(Review.BoundaryReview)
    writeText(fullfile(OutputFolder,'ResearcherBoundaryHistory.json'),jsonencode(Review.BoundaryReview.Document,'PrettyPrint',true));
end
writeText(fullfile(OutputFolder,'TimingReview.json'),jsonencode(Data.TimingReview,'PrettyPrint',true));
writeText(fullfile(OutputFolder,'BaselineDiagnostic.json'),jsonencode(Data.BaselineDiagnostic,'PrettyPrint',true));
if ~isempty(native)
    Receipt.NativeFrameMasks='attached_master_snapshot';
    Receipt.NativeMaskEvidence=rmfield(native,'FramePixels');
end
if isfield(Data,'WindowLink'),Receipt.WindowLink=Data.WindowLink;end
if isfield(Review,'AuditCreationReceipt'),Receipt.AuditCreationReceipt=Review.AuditCreationReceipt;end
Receipt.ReviewImplementation=struct('Function',{},'SHA256',{});
for name={'attachBOIReferenceSources','buildBOIReviewedReferencePreview','createBOIReviewedReferencePanel','getBOIBoundaryReview','loadBOIBoundaryReview','saveBOIBoundaryReview','validateBOIBoundaryAnnotation','createBOIBoundaryPanel','loadBOIEventReview','buildBOIEventReviewData','buildBOITimingReviewData','createBOITimingPanel','buildBOIBaselineDiagnostic','createBOIBaselinePanel','openBOIEventReview','exportBOIEventReview','getBOIMeasurementDictionary','attachBOINativeMasks','validateBOINativeMaskSource','linkBOIWindowEventReview'}
    Receipt.ReviewImplementation(end+1)=struct('Function',name{1},'SHA256',oxygenFileSHA256(which(name{1})));
end
% MAT preserves NaN and table types; JSON null means unavailable, never zero.
save(fullfile(OutputFolder,'SelectedEventReview.mat'),'Data','Receipt','ReviewedReferencePreview');
writeText(fullfile(OutputFolder,'SelectedEventReview.json'),jsonencode(Receipt,'PrettyPrint',true));
writeText(fullfile(OutputFolder,'README.txt'),sprintf(['Local BOI event review; scientific eligibility is not established.\n' ...
    'SelectedEventTrace.csv holds all recorded frames, explicit sample membership and signed fractions.\n' ...
    'TimingReview.json records corrected-primary, detection-support and raw correction-quality roles, saved-stage availability and provenance.\n' ...
    'CorrectedIntensity, FilteredDetectionScore and SavedSiteTimingTrace are saved signals; absent/invalid stages stay unavailable, not substituted.\n' ...
    'ExistingRemovedTrend is raw minus saved corrected mean only when their shared input is recorded; it is not a new fit or amplitude baseline.\n' ...
    'Timing sample positions use Frame; ModeledTimeSec=(Frame-1)/SampleHz. Exposure duration is separate. Manual judgments remain separate from saved measurement bounds.\n' ...
    'ResearcherBoundaries.json contains only SAVED judgments for this event; unsaved UI drafts are excluded.\n' ...
    'ResearcherBoundaryHistory.json, when attached, retains all reviewed events and old/new annotations, reviewer, reason and UTC revision time.\n' ...
    'Researcher alternatives are discrete one-based frames; empty endpoints are unresolved. No amplitude or native-mask recalculation uses them.\n' ...
    'ReviewedReferencePreview.json records proposed A/B eligibility only, not a new baseline mean or amplitude. Neither policy is adopted.\n' ...
    'When saved onsets exist, ReviewedReferenceFrames.csv and ReviewedReferenceContributors.csv explain sample eligibility and both-sign exclusions.\n' ...
    'Unknown native support stays NaN, not zero. Source hashes, saved revision, recovery contacts and alternatives accompany the preview.\n' ...
    'BaselineDiagnostic.json is a versioned review-only calculation; it never replaces the saved amplitude or mean baseline.\n' ...
    'Reference sensitivity is DifferenceFromAuditedAmplitude_pp: line minus the audited constant-reference amplitude on the same source.\n' ...
    'SavedMeasurementMatchesAudit and AuditAgreementMessage expose saved-versus-audited disagreement independently of trace replay.\n' ...
    'SavedBaseline and AuditedBaseline retain their distinct roles; missing historical StoredBaseline remains unavailable.\n' ...
    'DifferenceFromSavedAmplitude_pp is retained but may include audit disagreement; it is not the reference-sensitivity measure.\n' ...
    'DiagnosticLinearReference is present only on clean-baseline and measurement frames when a fit is available.\n' ...
    'DiagnosticLinearSignedFraction is present only on measurement frames when the whole alternative measurement is available.\n' ...
    'A nonpositive/nonfinite event reference withholds the entire alternative amplitude/integral; no clipping or earlier baseline search.\n' ...
    'Raw reference values may still be shown to explain an invalid extrapolation. Negative and very small values are retained.\n' ...
    'The original SignedFraction, selected saved event, dictionary and statistics are unchanged. MAT retains exact unavailable values and diagnostic metadata.\n' ...
    'CSV membership flags are 1 for included and 0 for excluded; use flag==1 when indexing after readtable.\n' ...
    'FixedAmplitudeFootprint.csv is the union used for amplitude, not a native frame mask or tissue boundary.\n' ...
    'If attached, NativeEventFramePixels.csv contains only the selected native event run.\n' ...
    'NativeMaskPixels in the trace CSV is NaN when unavailable, or a pixel count (including known zero outside the attached run).\n' ...
    'Native masks are detected event support, not reviewed anatomical tissue. Master association is recorded at attachment, not retroactively at the original audit.\n' ...
    'WindowLink, when present, identifies the originating statistics/window and event overlap; this association was checked when connecting the reviewers.\n' ...
    'Source image is not included. The audit contains saved quantitative traces; it is not a new analysis.\n' ...
    'NaN in MATLAB/CSV and null in JSON represent unavailable values, never zero.\n' ...
    'Dictionary role: %s. Selection alone does not establish the dictionary used by the audit.\n' ...
    'Original source paths and metadata are retained; review before public sharing.\n'],role));
files=dir(OutputFolder);manifest=struct('Name',{},'SHA256',{});
for k=1:numel(files)
    if ~files(k).isdir,manifest(end+1)=struct('Name',files(k).name,'SHA256',oxygenFileSHA256(fullfile(OutputFolder,files(k).name)));end %#ok<AGROW>
end
writeText(fullfile(OutputFolder,'ArtifactChecksums.json'),jsonencode(manifest,'PrettyPrint',true));
end
function writeText(path,value)
fid=fopen(path,'w');assert(fid>=0);cleanup=onCleanup(@()fclose(fid));fprintf(fid,'%s\n',value);
end
