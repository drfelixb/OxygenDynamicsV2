packet=fileparts(mfilename('fullpath'));workspace=fileparts(fileparts(packet));repo=fullfile(workspace,'existing-analysis');
addpath(repo);setupOxygenDynamicsPath;
first=jsondecode(fileread(fullfile(packet,'gui-integration.json')));
assert(first.OutputBytes<3*1024^3&&first.DetectorExecutions==1);
x=load(fullfile(packet,'gui-request.mat'));o=x.request.Options;o.OutputFolder=fullfile(packet,'batch-run');
pool=parpool('Threads',2);poolCleanup=onCleanup(@()delete(pool));
started=tic;request=prepareBOIRecordingRequest(o);Run=runBOIRecordingWorkflow(request);
assert(toc(started)<600,'Approved soft execution budget exceeded.');
files=dir(fullfile(o.OutputFolder,'**','*'));bytes=sum([files(~[files.isdir]).bytes]);
receipt=struct('Entry','batch service','RunManifest',Run.ManifestPath,'ElapsedSeconds',toc(started), ...
    'OutputBytes',bytes,'Events',Run.EventCount,'DetectorExecutions',1,'CombinedOutputBytes',bytes+first.OutputBytes);
writeBOIRunJSON(fullfile(packet,'batch-integration.json'),receipt);
assert(receipt.CombinedOutputBytes<=6*1024^3,'Approved combined output review limit exceeded.');
disp(receipt);
