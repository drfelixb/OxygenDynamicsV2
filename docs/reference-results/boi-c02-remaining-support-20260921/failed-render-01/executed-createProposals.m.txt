 end

 imwrite(Mask,fullfile(folder,'ProposedNativeMask.png'));

 fig=figure('Visible','off','Color','w','Position',[20 20 1500 1130]);cl=onCleanup(@()close(fig));

   j=mod(idx,size(vertices,1))+1;plot(ax,vertices([idx j],1),vertices([idx j],2),'-','Color',[1 .55 0],'LineWidth',1.8);
  end
  xlabel(ax,'Native column');ylabel(ax,'Native row');title(ax,sprintf('Frame %d | %g s',frames(k),frames(k)-1),'FontSize',12);
 end
 ax=axes(fig,'Position',[.71 .105 .26 .335]);axis(ax,'off');
 text(ax,0,1,sprintf('DRAFT ONLY — NOT ADOPTED\n\nSingle outline; no internal holes\nShared display [%g,%g]\nNo cross-session mask transfer\nNo anatomical reference established\n\nImage-edge segments bound only\nthe recorded field.\n\nDark interior tissue retained.\nRead the source-specific uncertainty\nnote before accepting the outline.',limits(1),limits(2)),'Units','normalized','VerticalAlignment','top','FontSize',12,'Interpreter','none');
