function P=exportBOITemporalContext(Review,Index,OutputFolder,Selection)
%EXPORTBOITEMPORALCONTEXT Save exact diagnostic ingredients in a new folder.
if nargin<4,Selection=[];end
assert(~isfolder(OutputFolder)&&~isfile(OutputFolder),'OxygenDynamics:ContextOutputExists','Choose a new context folder; prior evidence is preserved.');
P=buildBOITemporalContextPreview(Review,Index);P.DisplaySelection=Selection;
P.CreatedUTC=char(datetime('now','TimeZone','UTC','Format','yyyy-MM-dd''T''HH:mm:ssXXX'));
P.Implementation=struct('Function',{},'SHA256',{});
for n={'computeBOITemporalContext','buildBOITemporalContextPreview','buildBOITemporalNativeContacts','buildBOITemporalHumanContacts','createBOITemporalContextPanel','exportBOITemporalContext'}
 P.Implementation(end+1)=struct('Function',n{1},'SHA256',oxygenFileSHA256(which(n{1})));
end
mkdir(OutputFolder);write(fullfile(OutputFolder,'TemporalContext.json'),jsonencode(P,'PrettyPrint',true));save(fullfile(OutputFolder,'TemporalContext.mat'),'P');
Frame=(1:numel(P.CorrectedTrace))';CorrectedIntensity=P.CorrectedTrace;writetable(table(Frame,CorrectedIntensity),fullfile(OutputFolder,'CorrectedTrace.csv'));
PixelIndex=P.Footprint;writetable(table(PixelIndex),fullfile(OutputFolder,'FixedFootprint.csv'));
write(fullfile(OutputFolder,'README.txt'),sprintf(['Optional BOI temporal context, definition 1.0.0-diagnostic. No scientific rule adopted.\n' ...
 'Exact saved corrected-intensity samples on the original event-union footprint. External 1 Hz: modeled seconds=Frame-1. Exposure is separate.\n' ...
 'For interval a:b, before=a-W:a-1 and after=b+1:b+W, W=10 and 20. Endpoints belong to the interval.\n' ...
 'Each side requires all W observed finite samples. Median, min, max, range=max-min, unscaled MAD=median(abs(y-median(y))).\n' ...
 'Rise=max(interval)-side median; decline=side median-min(interval). Both signed values retained; no clipping or screen.\n' ...
 'Onset/recovery values, interval extrema with all tied frame IDs, endpoint difference and side-relative differences remain explicit.\n' ...
 'NaN/MAT and null/JSON mean unavailable with dependency statuses. No interpolation, shortening, fit or substitute baseline.\n' ...
 'Native contacts are both-sign union counts and contributors. NativeStatus=not_attached means unknown, not clean.\n' ...
 'Human contacts use current saved intervals and fixed-footprint intersections; approximate contextual observations stay separate.\n' ...
 'AcceptedReferences records exact loaded reference frames and source documents; diagnostic windows do not replace them or calculate a reference mean.\n' ...
 'BoundaryReview/BoundaryHistory preserve current alternatives, preferred marks, uncertainty and previous revisions; drafts are excluded.\n' ...
 'All rows/scales exported. DisplaySelection records the view, not an accepted outcome.\n' ...
 'Source hashes, pixel IDs, corrected samples and implementation hashes permit replay. Local paths are retained for internal traceability.\n']));
files=dir(OutputFolder);records={};for k=1:numel(files),if files(k).isdir,continue;end;f=fullfile(OutputFolder,files(k).name);records{end+1}=struct('Path',files(k).name,'Bytes',files(k).bytes,'SHA256',oxygenFileSHA256(f));end
write(fullfile(OutputFolder,'ArtifactChecksums.json'),jsonencode(records,'PrettyPrint',true));
end
function write(path,value)
f=fopen(path,'w');assert(f>=0);c=onCleanup(@()fclose(f));fprintf(f,'%s\n',value);
end
