root='../reference-validation/surge-pulse-onset-20260910';
old=readtable('../reference-validation/surge-onset-trace-panel-20260909/panel-results.csv','TextType','string','Delimiter',',');
new=readtable(fullfile(root,'panel-results.csv'),'TextType','string','Delimiter',',');
old=old(old.Context=="available",:);tables={old,new};A=zeros(2,3);B=zeros(3,2);rises=[5 15 30];
for j=1:2
 T=tables{j};rows=logical(T.Constructible);resolved=rows&T.Status=="resolved";good=resolved&T.WithinFiveSec;
 A(j,:)=100*[nnz(good) nnz(resolved&~good) nnz(rows&~resolved)]/nnz(rows);
 for k=1:3,selected=rows&T.RiseFrames==rises(k);B(k,j)=100*nnz(selected&T.WithinFiveSec)/nnz(selected);end
end
F=figure('Visible','off','Position',[100 100 1100 520],'Color','white');tiledlayout(1,2,'Padding','compact','TileSpacing','compact');
nexttile;h=bar(A,'stacked');h(1).FaceColor=[.15 .48 .64];h(2).FaceColor=[.86 .57 .23];h(3).FaceColor=[.83 .85 .87];
xticklabels({'Rising fit','Pulse fit'});ylim([0 100]);ylabel('% of constructible traces');title('All onset outcomes');
legend({'Within 5 s','Resolved, outside 5 s','Unresolved'},'Location','southoutside');box off;
nexttile;h=bar(B);h(1).FaceColor=[.5 .56 .60];h(2).FaceColor=[.15 .48 .64];
xticklabels({'5 s','15 s','30 s'});xlabel('Imposed rise length');ylabel('% within 5 s of known start');ylim([0 35]);
title('Benefits depend on rise length');legend({'Rising fit','Pulse fit'},'Location','southoutside');box off;
sgtitle({'Conditional onset comparison on archived backgrounds', ...
 '5,136 traces per method; three BOI recordings + one fluorescence control'},'FontWeight','normal','Color',[.12 .12 .12]);
set(findall(F,'Type','axes'),'Color','white','XColor',[.12 .12 .12],'YColor',[.12 .12 .12]);
set(findall(F,'Type','text'),'Color',[.12 .12 .12]);set(findall(F,'Type','legend'),'Color','white','TextColor',[.12 .12 .12],'EdgeColor','white');
exportgraphics(F,fullfile(root,'pulse-comparison.png'),'Resolution',160,'BackgroundColor','white');close(F);
for path=["tests/analysis/estimateSurgePulseOnset.m","tests/analysis/runSurgePulseOnsetPanel.m","tests/analysis/testSurgePulseOnset.m"]
 disp(path);disp(checkcode(path,'-id'));
end
