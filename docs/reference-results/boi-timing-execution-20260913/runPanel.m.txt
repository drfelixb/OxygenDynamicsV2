function x=baseRow(r,e,a,role,lo,hi)
x=struct('RecordingKey',string(r.RecordingKey),'EventType',string(e.EventType),'SiteID',e.SiteID,'EventID',e.EventID,'AuditRow',e.AuditRow,'Role',role,'NativeStart',a.DetectedStartFrame,'NativeEnd',a.DetectedEndFrame,'SavedStart',a.StartFrame,'SavedEnd',a.EndFrame,'StartFrame',lo,'EndFrame',hi,'MarkedTimeSeparationSec',hi-lo,'InclusiveDurationSec',hi-lo+1,'SearchLower',NaN,'SearchUpper',NaN,'PreviousNativeEnd',NaN,'NextNativeStart',NaN,'SeedFrame',NaN,'OnsetBefore',NaN,'OnsetAfter',NaN,'RecoveryBefore',NaN,'RecoveryAfter',NaN,'OnsetCensor',"",'RecoveryCensor',"",'TimingStatus',"saved_bounds",'CoversWholeNative',false,'NativeSignInterruptions',NaN,'BaselineStart',NaN,'BaselineEnd',NaN,'CleanSamples',NaN,'OverlapExcludedSamples',NaN,'NonfiniteBaselineSamples',NaN,'BaselineMean',NaN,'Amplitude',NaN,'AmplitudeStatus',"not_calculated",'SavedAmplitude',a.StoredAmplitude,'SavedBaseline',a.StoredBaseline,'AmplitudeDifferencePP',NaN);
end
function m=measure(raw,overlap,lo,hi,sign)
b=(max(1,lo-20):lo-1)';clean=b(~overlap(b)&isfinite(raw(b)));m=struct('BaselineStart',lo-20,'BaselineEnd',lo-1,'CleanSamples',numel(clean),'OverlapExcludedSamples',nnz(overlap(b)),'NonfiniteBaselineSamples',nnz(~isfinite(raw(b))),'BaselineMean',NaN,'Amplitude',NaN,'AmplitudeStatus',"insufficient_clean_prebaseline",'CleanFrames',clean);
if numel(clean)==20
 B=mean(raw(clean));m.BaselineMean=B;if isfinite(B)&&B>0&&all(isfinite(raw(lo:hi))),d=(raw(lo:hi)-B)/B;if strcmp(sign,'sink'),m.Amplitude=-min(d);else,m.Amplitude=max(d);end;m.AmplitudeStatus="valid";else,m.AmplitudeStatus="nonpositive_baseline_or_missing_event_signal";end
end
end
function row=joinMeasurement(row,m)
fields=setdiff(fieldnames(m),{'CleanFrames'});for k=1:numel(fields),row.(fields{k})=m.(fields{k});end;row.AmplitudeDifferencePP=100*(row.Amplitude-row.SavedAmplitude);
end
function reason=boundaryReason(edge,bound,limit,acq,neighbor,native,isLeft)
reason="";if edge~=bound,return;end
if edge==acq,reason=reason+"acquisition_boundary;";end
if edge==limit,reason=reason+"30_frame_search_limit;";end
if isfinite(neighbor)
 midpoint=floor((neighbor+native)/2);if isLeft,midpoint=midpoint+1;end
 if edge==midpoint,reason=reason+"same_site_neighbor_partition;";end
end
end
function same=sameNumber(a,b)
same=(isnan(a)&&isnan(b))||(isfinite(a)&&isfinite(b)&&abs(a-b)<=1e-9+1e-12*abs(b));
end
function p=unionIndices(a,b),p=union(a,b);end
function writeJson(path,value)
fid=fopen(path,'w');assert(fid>=0);c=onCleanup(@()fclose(fid));fprintf(fid,'%s\n',jsonencode(value,'PrettyPrint',true));
end
