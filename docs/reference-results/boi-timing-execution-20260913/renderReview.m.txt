root=fileparts(mfilename('fullpath'));queue=jsondecode(fileread(fullfile(root,'initial-review-queue.json')));qc=queue.Cases;if isstruct(qc),qc=num2cell(qc);end
out=fullfile(root,'review');assert(~isfolder(out));mkdir(out);
for q=1:numel(qc)
 e=qc{q};r=struct('RecordingKey',e.RecordingKey);name=sprintf('%s-site%d-event%d',e.EventType,e.SiteID,e.EventID);source=fullfile(root,'run-02',r.RecordingKey,name);folder=fullfile(out,r.RecordingKey,name);mkdir(folder);
 T=readtable(fullfile(source,'trace.csv'));W=readtable(fullfile(root,'checked-tables',[r.RecordingKey '.csv']));W=W(string(W.EventType)==string(e.EventType)&W.SiteID==e.SiteID&W.EventID==e.EventID,:);saved=W(string(W.Role)=="saved",:);
 a=struct('StartFrame',saved.StartFrame,'EndFrame',saved.EndFrame,'DetectedStartFrame',saved.NativeStart,'DetectedEndFrame',saved.NativeEnd,'StoredStatus',string(saved.AmplitudeStatus));branches={table2struct(W(string(W.Role)=="negative",:)),table2struct(W(string(W.Role)=="positive",:))};lower=branches{1}.SearchLower;upper=branches{1}.SearchUpper;
 M=load(fullfile(source,'native-masks.mat'));native=(a.DetectedStartFrame:a.DetectedEndFrame)';c=cell(height(T),1);c(native)=M.NativeFramePixels;F=readtable(fullfile(source,'footprint.csv'));px=F.PixelIndexMATLAB;
 plotReview(folder,r,e,T,a,branches,lower,upper,c,native,px);
end
function plotReview(folder,r,e,T,a,branches,lower,upper,c,native,px)
f=figure('Visible','off','Color','w','Position',[20 20 1300 1030]);cl=onCleanup(@()delete(f));layout=tiledlayout(f,3,1,'TileSpacing','compact','Padding','compact');x=T.Frame;show=x>=max(1,min([lower,a.StartFrame-20])-10)&x<=min(height(T),max(upper,a.EndFrame)+10);
for panel=1:3
 ax=nexttile(layout);set(ax,'Color','w','XColor',[.15 .15 .15],'YColor',[.15 .15 .15],'FontSize',11);hold(ax,'on');
 if panel==1
  plot(ax,x(show),T.RawMean(show),'Color',[0 .3 .65]);plot(ax,x(show),T.ExistingRemovedTrend(show),'Color',[.85 .4 0],'LineWidth',1.4);plot(ax,x(T.OriginalCleanBaseline&show),T.RawMean(T.OriginalCleanBaseline&show),'.','Color',[0 .55 .2],'MarkerSize',11);names={'Preserved source','Existing cubic trend'};if any(T.OriginalCleanBaseline&show),names{3}='Saved clean baseline';end;lg=legend(ax,names,'Location','southoutside','NumColumns',numel(names));ylabel(ax,'Input intensity (a.u.)','Color','k');
 elseif panel==2
  plot(ax,x(show),T.CubicCorrectedMean(show),'Color',[0 .3 .65]);yline(ax,0,':','HandleVisibility','off','Color',[.4 .4 .4]);lg=legend(ax,{'Existing corrected local source'},'Location','southoutside');ylabel(ax,'Corrected intensity','Color','k');
  colors={[.55 .15 .65],[0 .5 .4]};for b=1:2,z=branches{b};if isfinite(z.SeedFrame),for v=[z.StartFrame,z.EndFrame],xline(ax,v,'-.','Color',colors{b},'LineWidth',1.1,'HandleVisibility','off');end;plot(ax,z.SeedFrame,T.CubicCorrectedMean(z.SeedFrame),'o','Color',colors{b},'HandleVisibility','off');end,end
  title(ax,sprintf('Purple: negative run (%s); green: positive run (%s)',plainStatus(branches{1}.TimingStatus),plainStatus(branches{2}.TimingStatus)),'Interpreter','none','Color','k','FontSize',10);
 else
  plot(ax,x(show),T.FilteredMean(show),'Color',[.85 .4 0]);plot(ax,x(show),T.SavedSiteTimingTrace(show),':','Color',[.3 .3 .3]);lg=legend(ax,{'Fixed-footprint filtered score','Saved site timing trace'},'Location','southoutside','NumColumns',2);ylabel(ax,'Detection score','Color','k');xlabel(ax,'Recording frame (external trigger: 1 Hz)','Color','k');
 end
 set(lg,'TextColor','k','Color','w','EdgeColor',[.8 .8 .8]);for v=[a.DetectedStartFrame,a.DetectedEndFrame],xline(ax,v,'k--','LineWidth',1.2,'HandleVisibility','off');end;for v=[a.StartFrame,a.EndFrame],xline(ax,v,':','Color',[.5 .5 .5],'HandleVisibility','off');end
 for v=[lower,upper],xline(ax,v,':','Color',[.7 .7 .7],'HandleVisibility','off');end
 if strcmp(r.RecordingKey,'FB2314-awake')&&strcmp(e.EventType,'surge')&&e.SiteID==1&&ismember(e.EventID,[3 4])
  marks=[496 498 514 516];if e.EventID==4,marks=[536 538 554 556];end;for v=marks,xline(ax,v,':','Color',[.2 .4 .85],'HandleVisibility','off');end
 end
 xlim(ax,[min(x(show)),max(x(show))]);grid(ax,'on');ax.GridColor=[.7 .7 .7];
end
title(layout,{sprintf('%s | %s site %d event %d | review only',r.RecordingKey,e.EventType,e.SiteID,e.EventID),sprintf('Black: native; gray: saved/search | saved amplitude %s | diagnostic only',plainStatus(a.StoredStatus))},'Interpreter','none','Color','k');exportgraphics(f,fullfile(folder,'timing-review.png'),'Resolution',110,'BackgroundColor','white');clear cl
f=figure('Visible','off','Color','w','Position',[10 10 1200 430]);cl=onCleanup(@()delete(f));layout=tiledlayout(f,1,3,'Padding','compact');ff=unique([native(1),native(ceil(numel(native)/2)),native(end)],'stable');foot=false(512);foot(px)=true;
for q=1:numel(ff),ax=nexttile(layout);mask=false(512);mask(c{ff(q)})=true;imagesc(ax,mask,[0 1]);colormap(ax,gray);axis(ax,'image');set(ax,'XColor','k','YColor','k');hold(ax,'on');contour(ax,foot,[.5 .5],'Color',[.9 .4 0]);title(ax,sprintf('Native frame %d (%d pixels)',ff(q),nnz(mask)),'Color','k');end
title(layout,'Saved native masks, source coordinates | orange: fixed event union | no anatomical inference','Color','k');exportgraphics(f,fullfile(folder,'native-masks.png'),'Resolution',100,'BackgroundColor','white');clear cl
end

function text=plainStatus(status)
text=string(status);switch text
case 'fully_bracketed_diagnostic',text='two crossings';case 'censored_diagnostic',text='boundary limited';case 'no_native_excursion_of_requested_sign',text='absent inside native interval';case 'insufficient_clean_prebaseline',text='unavailable: insufficient baseline';case 'valid',text='available';end
end
