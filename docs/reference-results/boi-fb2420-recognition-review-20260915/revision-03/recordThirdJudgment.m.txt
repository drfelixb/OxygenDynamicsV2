function recordThirdJudgment
here=fileparts(mfilename('fullpath'));root=fileparts(fileparts(fileparts(here)));
addpath(fullfile(root,'existing-analysis'));setupOxygenDynamicsPath;
f=fullfile(here,'feedback.json');d=jsondecode(fileread(f));s=d.SourceBindings;names=fieldnames(s);
for k=1:numel(names),v=s.(names{k});assert(strcmp(oxygenFileSHA256(v.Path),v.SHA256));end
R=loadBOIEventReview(s.Audit.Path);Previous=loadBOIBoundaryReview(R,s.PreviousReview.Path);
A=struct('Status','not_recognized','OnsetFrames',[],'RecoveryFrames',[], ...
 'PreferredOnsetFrame',[],'PreferredRecoveryFrame',[],'Reviewer','Researcher (user)', ...
 'Reason',['Automatic frames 2-4 judged as fluctuation in subsequent-trace context. Researcher separately identifies a pocket starting around frame 7 until likely frame 27 on this footprint trace; approximate timing is not assigned to this native event. Exact reply/context SHA256: ' oxygenFileSHA256(f)]);
newpath=fullfile(here,'BOI-researcher-review-03.json');
Store=saveBOIBoundaryReview(R,22,A,newpath,Previous);
assert(numel(Store.Document.Revisions)==3);
assert(isequaln(Store.Document.Revisions(1:2),Previous.Document.Revisions));
R.BoundaryReview=Store;B=getBOIBoundaryReview(R,22);
assert(strcmp(B.Status,'not_recognized')&&isempty(B.Annotation.OnsetFrames)&&isempty(B.Annotation.RecoveryFrames));
assert(strcmp(getBOIBoundaryReview(R,34).Status,'not_recognized'));
assert(strcmp(getBOIBoundaryReview(R,106).Status,'not_reviewed'));
addpath(fullfile(root,'reference-validation','boi-fb2420-first-roi-run-20260915'));
for pos=[3 4]
 [Fig,UI]=openFB2420Review(pos);UI.LoadBoundaries(newpath);drawnow;
 exportapp(Fig,fullfile(here,sprintf('case-%02d-timing.png',pos)));close(Fig);
end
for k=1:numel(names),v=s.(names{k});assert(strcmp(oxygenFileSHA256(v.Path),v.SHA256));end
v=struct('Status','saved_reloaded_and_rendered','CumulativeRevisionCount',3, ...
 'AnnotationAuditRow',22,'AnnotationStatus','not_recognized','NoBoundaryFrames',true, ...
 'ApproximateContextObservationRemainsSeparate',true,'PreviousRevisionPreserved',true, ...
 'AutomaticSourcesUnchanged',true,'NextQueuePosition',4,'NextAuditRow',106, ...
 'NextStatus','not_reviewed','BoundaryArtifact',struct('Path',Store.Path,'SHA256',Store.SHA256));
fid=fopen(fullfile(here,'verification.json'),'w');fprintf(fid,'%s\n',jsonencode(v,'PrettyPrint',true));fclose(fid);
end
