function plotSourceInspection
here=fileparts(mfilename('fullpath'));out=fullfile(here,'run-01');
L=jsondecode(fileread(fullfile(out,'display-layout.json')));
fid=fopen(fullfile(out,'display-arrays.bin'),'r','ieee-le');assert(fid>=0);
A=fread(fid,inf,'double');fclose(fid);assert(numel(A)==prod(L.Shape));
A=permute(reshape(A,[512 512 11]),[2 1 3]);
Q=jsondecode(fileread(fullfile(out,'source-qc.json')));
assert(max(abs(A(:,:,1)-round(A(:,:,1))),[],'all')==0);
fig=figure('Visible','off','Color','w','Position',[50 50 1350 850]);
tiledlayout(fig,2,3,'TileSpacing','compact','Padding','compact');
for k=1:6
    ax=nexttile;show(ax,A(:,:,k+5),sprintf('Frame %d | modeled %d s',Q.SnapshotFrames(k),Q.SnapshotFrames(k)-1),L.SourceDisplayLimits);
end
sgtitle('FB2420 preserved BOI source | shared display scale; no crop, mask or registration');
exportgraphics(fig,fullfile(out,'source-six-frames.png'),'Resolution',140);close(fig);
fig=figure('Visible','off','Color','w','Position',[50 50 1100 1000]);
tiledlayout(fig,2,2,'TileSpacing','compact','Padding','compact');
ax=nexttile;show(ax,A(:,:,1),{'490 nm after reference candidate','Own display scale; transform unverified'},L.ReferenceDisplayLimits);colorbar(ax);
names={'BOI mean: frames 1-30','BOI mean: frames 586-615','BOI mean: frames 1171-1200'};
for k=1:3,ax=nexttile;show(ax,A(:,:,k+2),names{k},L.SourceDisplayLimits);colorbar(ax);end
sgtitle({'FB2420 reference and source in native grids','Same shape does not prove spatial correspondence | colorbars in input units'});
exportgraphics(fig,fullfile(out,'reference-source-comparison.png'),'Resolution',140);
exportgraphics(fig,fullfile(out,'reference-source-comparison.pdf'),'ContentType','vector');close(fig);
T=readtable(fullfile(out,'frame-qc.csv'));
fig=figure('Visible','off','Color','w','Position',[50 50 1200 760]);tiledlayout(fig,3,1,'TileSpacing','compact');
ax=nexttile;plot(ax,T.Frame,T.WholeFieldMean);ylabel('Whole-field mean');grid on;xlim([1 1200]);
ax=nexttile;plot(ax,T.Frame,[T.Minimum T.Maximum]);ylabel('Input intensity');legend('Minimum','Maximum');grid on;xlim([1 1200]);
ax=nexttile;plot(ax,T.Frame,[T.ZeroPixels T.Uint16MaximumPixels]);ylabel('Pixels / frame');legend('Zero pixels','65535 pixels');grid on;xlim([1 1200]);xlabel('Recording frame (external 1 Hz; frame 1 = modeled 0 s)');
sgtitle('FB2420 whole-field source QC | not an event trace or physiological baseline');
exportgraphics(fig,fullfile(out,'whole-field-qc.png'),'Resolution',140);close(fig);
disp('Three source inspection figures exported; no mask or registration adopted.');
end
function show(ax,a,name,limits)
imagesc(ax,1:512,1:512,a,limits');axis(ax,'image');set(ax,'YDir','reverse');colormap(ax,gray(256));
title(ax,name);xlabel(ax,'Native column (1-based)');ylabel(ax,'Native row (1-based)');
end
