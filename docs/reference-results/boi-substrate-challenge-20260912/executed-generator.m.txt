function h=hashArray(A)
 md=java.security.MessageDigest.getInstance('SHA-256');
 for f=1:size(A,3),v=A(:,:,f);md.update(typecast(v(:),'uint8'));end
 h=lower(reshape(dec2hex(typecast(md.digest(),'uint8'),2)',1,[]));
end
function writeJSON(p,v)
 fid=fopen(p,'w');assert(fid>0);clean=onCleanup(@()fclose(fid));fprintf(fid,'%s\n',jsonencode(v,'PrettyPrint',true));
end
function plotBaseline(T,p)

  nexttile;hold on;hl=[Inf,600,1200,2400];sel=T.HalfLifeSec==hl(k)&T.NoiseFraction==0;
  colors=lines(3);for j=1:3,dur=[5,20,60];r=sel&T.DurationSec==dur(j);plot(100*T.LocalFraction(r),100*T.AmplitudeFraction(r),'-o','Color',colors(j,:),'DisplayName',sprintf('%d s event',dur(j)));end
  xline(0,':');yline(0,':');grid on;xlabel('Injected signed local change (%)');ylabel('Reported sign-specific amplitude (%)');title(sprintf('Illustrative half-life: %g s',hl(k)));legend('Location','best');
 end
 sgtitle('Fixed 20-second raw baseline: known event support, noise-free fixtures');exportgraphics(f,p,'Resolution',140);close(f);
end
