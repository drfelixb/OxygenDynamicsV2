function verifyRevision04
root='/Users/zcm361/Documents/Github/OxygenDynamicsV2';cd(fullfile(root,'existing-analysis'));setupOxygenDynamicsPath;
out=fullfile(root,'reference-validation/boi-researcher-boundary-walkthrough-20260914/revision-04');I=jsondecode(fileread(fullfile(out,'input.json')));
R=loadBOIEventReview(fullfile(fileparts(I.ReviewPath),'event-amplitude-audit.mat'));
old=loadBOIBoundaryReview(R,I.PriorReviewPath);new=loadBOIBoundaryReview(R,I.ReviewPath);
assert(strcmp(new.SHA256,I.SHA256)&&strcmp(old.SHA256,I.PriorSHA256));
assert(numel(new.Document.Revisions)==4&&isequal(new.Document.Revisions(1:3),old.Document.Revisions));
assert(strcmp(new.Document.PreviousArtifact.SHA256,old.SHA256));
reviewed=R;reviewed.BoundaryReview=new;rows=[309 308 321 186];
for row=rows
    before=buildBOIEventReviewData(R,row);after=buildBOIEventReviewData(reviewed,row);
    assert(isequaln(rmfield(before,'ResearcherBoundaryReview'),rmfield(after,'ResearcherBoundaryReview')));
    A=after.ResearcherBoundaryReview.Annotation;assert(strcmp(A.Status,'uncertain')&&strcmp(A.Reviewer,'Felix'));
end
B=getBOIBoundaryReview(reviewed,186);A=B.Annotation;
assert(isequal(A.OnsetFrames,[1149 1153])&&isequal(A.RecoveryFrames,[1166 1167])&&A.PreferredOnsetFrame==1153&&isempty(A.PreferredRecoveryFrame));
assert(strcmp(oxygenFileSHA256(R.AuditPath),R.AuditSHA256)&&strcmp(oxygenFileSHA256(I.ReviewPath),I.SHA256)&&strcmp(oxygenFileSHA256(I.PriorReviewPath),I.PriorSHA256));
V=struct('Status','saved_file_validation_passed','VerifiedUTC',char(datetime('now','TimeZone','UTC','Format','yyyy-MM-dd''T''HH:mm:ss''Z''')), ...
    'ReviewSHA256',new.SHA256,'PriorReviewSHA256',old.SHA256,'AuditSHA256',R.AuditSHA256, ...
    'RevisionCount',4,'AuditRows',rows,'EarlierRevisionsUnchanged',true,'LatestAnnotation',A, ...
    'SavedPreferredOnsetFrame',1153,'PreviouslySpecifiedPreferredOnsetFrame',1154, ...
    'SavedPreferredOnsetModeledTimeSec',1152,'EarlierPreferredOnsetModeledTimeSec',1153, ...
    'ClarificationStatusAtVerification','requested; no inferred correction', ...
    'PriorMeasurementFieldsUnchanged',true,'OriginalFilesUnchanged',true,'UserReportedSave',true, ...
    'ReopenedBy','assistant MATLAB loader','ScientificAcceptance','not_established');
fid=fopen(fullfile(out,'verification.json'),'w');cleanup=onCleanup(@()fclose(fid));fprintf(fid,'%s\n',jsonencode(V,'PrettyPrint',true));disp(V);
end
