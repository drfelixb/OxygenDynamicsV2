function a=getAmp(q,kind)
 if kind=="sink",a=-min(q);else,a=max(q);end
end
function h=hashArray(A)
 md=java.security.MessageDigest.getInstance('SHA-256');for f=1:size(A,3),v=A(:,:,f);md.update(typecast(v(:),'uint8'));end;h=lower(reshape(dec2hex(typecast(md.digest(),'uint8'),2)',1,[]));
end
function writeJSON(p,v)
 fid=fopen(p,'w');assert(fid>0);clean=onCleanup(@()fclose(fid));fprintf(fid,'%s\n',jsonencode(v,'PrettyPrint',true));
end
function plotErrors(T,p)
 fig=figure('Visible','off','Position',[0,0,1280,980]);tiledlayout(2,2,'TileSpacing','compact');arms=["smooth","fluctuating"];colors=lines(4);
 for ai=1:2
  for ns=[0,.005]
   nexttile;hold on;A=T(T.Arm==arms(ai)&T.NoiseFraction==ns&T.Reference=="constant_20s",:);B=T(T.Arm==arms(ai)&T.NoiseFraction==ns&T.Reference=="linear_20s",:);assert(isequal(A.CaseID,B.CaseID));
   if ai==1,groups=[5,20,60];col='DurationSec';labels={'5-s event','20-s event','60-s event'};else,groups=[0,pi/2,pi,3*pi/2];col='PhaseRadians';labels={'Phase 0','Phase pi/2','Phase pi','Phase 3pi/2'};end
   xa=100*abs(A.AmplitudeErrorVsInjectedFraction);yb=100*abs(B.AmplitudeErrorVsInjectedFraction);h=gobjects(numel(groups),1);
   for j=1:numel(groups),sel=A.(col)==groups(j);h(j)=scatter(xa(sel),yb(sel),30,colors(j,:),'filled','MarkerFaceAlpha',.65);end
   limit=max([xa;yb])*1.07;if limit==0,limit=1;end;plot([0,limit],[0,limit],'k:','HandleVisibility','off');xlim([0,limit]);ylim([0,limit]);axis square;grid on;legend(h,labels,'Location','best');xlabel('20-s mean: absolute amplitude error (pp)');ylabel('20-s line: absolute amplitude error (pp)');title(sprintf('%s background; noise SD %.1f%% of initial signal',arms(ai),100*ns));
  end
 end
 sgtitle({'Error against the injected local component','Below diagonal: smaller error with line; synthetic sensitivity, no automatic method choice'});exportgraphics(fig,p,'Resolution',140);close(fig);
end
function plotPhases(C,Traces,BG,S,p)

  ci=find(C.Arm=="fluctuating"&isinf(C.HalfLifeSec)&C.NoiseFraction==0&C.LocalFraction==0&C.Kind=="sink"&C.PhaseRadians==phases(j));assert(isscalar(ci));U=S(S.CaseID==C.CaseID(ci),:);B=mean(Traces(ci,280:299));t=U.Frame-300;nexttile;hold on;
